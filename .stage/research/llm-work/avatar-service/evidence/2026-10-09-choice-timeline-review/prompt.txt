   3: function timestamp(value) {
   4:   if (typeof value !== 'string') return null;
   5:   const match = TIMESTAMP.exec(value);
   6:   if (!match) return null;
   7:   const [, yearText, monthText, dayText, hourText, minuteText, secondText, zone] = match;
   8:   const [year, month, day, hour, minute, second] =
   9:     [yearText, monthText, dayText, hourText, minuteText, secondText].map(Number);
  10:   const lastDay = new Date(Date.UTC(year, month, 0)).getUTCDate();
  11:   if (month < 1 || month > 12 || day < 1 || day > lastDay || hour > 23
  12:       || minute > 59 || second > 59) return null;
  13:   if (zone !== 'Z') {
  14:     const zoneHour = Number(zone.slice(1, 3));
  15:     const zoneMinute = Number(zone.slice(4));
  16:     if (zoneHour > 14 || zoneMinute > 59 || (zoneHour === 14 && zoneMinute !== 0)) return null;
  17:   }
  18:   const valueMs = Date.parse(value);
  19:   return Number.isFinite(valueMs) ? valueMs : null;
  20: }

  22: const isoDay = date => date.toISOString().slice(0, 10);

  24: function weekFor(day) {
  25:   const date = new Date(day + 'T00:00:00Z');
  26:   date.setUTCDate(date.getUTCDate() - (date.getUTCDay() + 6) % 7);
  27:   const start = isoDay(date);
  28:   date.setUTCDate(date.getUTCDate() + 6);
  29:   return {start, end: isoDay(date)};
  30: }

  32: export function groupChoiceTimeline(choices, timeZone) {
  33:   const formatter = new Intl.DateTimeFormat('en-GB', {
  34:     timeZone, calendar: 'gregory', year: 'numeric', month: '2-digit', day: '2-digit',
  35:   });
  36:   const days = new Map();
  37:   const weeks = new Map();
  38:   const dayForChoice = new Map();
  39:   let omittedCount = 0;
  40:   const dated = choices.flatMap((choice, index) => {
  41:     const at = timestamp(choice.recorded_at);
  42:     if (at === null) { omittedCount++; return []; }
  43:     return [{choice, at, index}];
  44:   }).sort((a, b) => a.at - b.at || a.index - b.index);
  45: 
  46:   for (const {choice, at} of dated) {
  47:     const parts = Object.fromEntries(formatter.formatToParts(new Date(at))
  48:       .filter(part => ['year', 'month', 'day'].includes(part.type))
  49:       .map(part => [part.type, part.value]));
  50:     const day = `${parts.year}-${parts.month}-${parts.day}`;
  51:     dayForChoice.set(choice, day);
  52:     if (!days.has(day)) days.set(day, {start: day, choices: []});
  53:     days.get(day).choices.push(choice);
  54: 
  55:     const week = weekFor(day);
  56:     if (!weeks.has(week.start)) weeks.set(week.start, {...week, choices: []});
  57:     weeks.get(week.start).choices.push(choice);
  58:   }
  59:   const newestFirst = groups => [...groups.values()]
  60:     .sort((a, b) => b.start.localeCompare(a.start));
  61:   return {days: newestFirst(days), weeks: newestFirst(weeks), dayForChoice, omittedCount};
  62: }
